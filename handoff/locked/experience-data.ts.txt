  );
}

const SIGNAL_LABELS: Record<AxisKey, [string, string]> = {
  future: ["점진적 미래설계", "적극적 미래설계"],
  growth: ["분배·안전망", "성장·활력"],
  universal: ["선별 집중", "보편 확대"],
};

export function getProgramInsight(program: BudgetProgram) {
  const base = getFieldImpact(program.fieldId);

  const signals = (Object.entries(program.tags) as Array<[AxisKey, number]>)
    .filter(([, value]) => value !== 0)
    .sort((a, b) => Math.abs(b[1]) - Math.abs(a[1]))
    .slice(0, 2)
    .map(([axis, value]) => SIGNAL_LABELS[axis][value > 0 ? 1 : 0]);

  return {
    ...base,
    signals,
    citizenSummary: `${program.name}에 1만 원을 배정하면 전체 실험 예산의 10%를 이 선택에 쓰는 셈입니다.`,
    opportunityCost:
      "이 금액을 선택하면 다른 프로그램 하나에 배정할 수 있는 1만 원의 기회를 포기합니다.",
  };
}
