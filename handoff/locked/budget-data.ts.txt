export function getProgramTags(field: BudgetField, name: string): AxisProfile {
  const tags = { ...field.profile };
  for (const rule of KEYWORD_TAGS) {
    if (rule.words.some((word) => name.includes(word))) {
      tags[rule.axis] = clamp(tags[rule.axis] + rule.value);
    }
  }
  return tags;
}

const isHiddenProgram = (name: string) =>
  HIDDEN_PROGRAM_NAMES.includes(name) ||
  HIDDEN_PROGRAM_NAME_PARTS.some((part) => name.includes(part));

/** 남길 분야가 정해진 사업인데 다른 분야에 붙어 있는 것. */
const isDisplacedProgram = (name: string, fieldId: string) => {
  const home = PROGRAM_HOME_FIELD[name];
  return home !== undefined && home !== fieldId;
};

export const HIDDEN_PROGRAMS = OFFICIAL_BUDGET_PROGRAMS.filter((program) =>
  isHiddenProgram(program.name),
);

const UNMERGED_PROGRAMS: BudgetProgram[] = OFFICIAL_BUDGET_PROGRAMS.filter(
  (program) =>
    !isHiddenProgram(program.name) &&
    !isDisplacedProgram(program.name, program.fieldId),
).map(
  (program) => {
    const field = FIELDS.find((candidate) => candidate.id === program.fieldId);
    if (!field) {
      throw new Error(`Unknown budget field: ${program.fieldId}`);
    }
    const unitNames = program.units.map((unit) => unit.name);
    return {
      id: program.id,
      fieldId: program.fieldId,
      fieldName: program.fieldName,
      // 원본 이름은 아래 officialName에 그대로 남습니다.
      name: PROGRAM_RENAMES[program.id] ?? program.name,
      // 화면에는 소관부처만 보여줍니다. 부문(sectors)까지 같이 늘어놓으면
      // 어디까지가 부처인지 구분되지 않아 오히려 읽기 어려웠습니다.
      description: program.owners.join(" · "),
      officialName: program.name,
      includedPrograms: unitNames,
      owners: program.owners,
      sectors: program.sectors,
      units: program.units,
      tags:
        BUDGET_GROUP_TAGS[program.fieldId]?.[program.name] ??
        getProgramTags(field, [program.name, ...unitNames].join(" ")),
    };
  },
);

/**
 * PROGRAM_MERGES를 적용합니다. 대표 프로그램이 흡수 대상의 단위사업과 소관부처를
 * 받아 가고, 흡수된 쪽은 목록에서 빠집니다. 대표의 id와 officialName은 그대로
 * 두어 묶음 배치(program-groups.ts는 원본 이름으로 적혀 있습니다)가 유지됩니다.
 */
const mergedAwayIds = new Set(PROGRAM_MERGES.flatMap((merge) => merge.absorb));

export const PROGRAMS: BudgetProgram[] = UNMERGED_PROGRAMS.filter(
  (program) => !mergedAwayIds.has(program.id),
).map((program) => {
  const merge = PROGRAM_MERGES.find((entry) => entry.keep === program.id);
  if (!merge) return program;

  const absorbed = merge.absorb
    .map((id) => UNMERGED_PROGRAMS.find((entry) => entry.id === id))
    .filter((entry): entry is BudgetProgram => Boolean(entry));
  const parts = [program, ...absorbed];
  const unique = (values: string[]) => [...new Set(values)];

  return {
    ...program,
    name: merge.name,
    owners: unique(parts.flatMap((part) => part.owners)),
    description: unique(parts.flatMap((part) => part.owners)).join(" · "),
    sectors: unique(parts.flatMap((part) => part.sectors)),
    units: parts.flatMap((part) => part.units),
    includedPrograms: parts.flatMap((part) => part.includedPrograms),
    mergedFrom: parts.map((part) => part.officialName),
  };
});

export const SELECTABLE_PROGRAMS = PROGRAMS.filter(
  (program) =>
    FIELDS.find((field) => field.id === program.fieldId)?.selectable !== false,
);

/**

 * 범주는 목록을 접고 펴서 찾기 쉽게 하는 껍데기로만 씁니다. 분야 하나가
 * 프로그램 40여 개를 한꺼번에 쏟아내던 것을 묶음 9개 정도로 줄입니다.

/**
 * 설명은 대개 "무엇인지" 한 문장 뒤에 "무엇을 포함하는지·옆 묶음과 뭐가 다른지"가
 * 이어집니다. 닫힌 목록에서는 첫 문장이면 어느 묶음인지 가려낼 수 있어, 거기까지만
 * 잘라 씁니다(126개 기준 평균 29자·최대 48자).
 */
const firstSentence = (description: string) =>
  description.split(/(?<=다\.)\s/)[0] ?? description;

/**
 * 분류 대기 칸. program-groups-extra.ts가 채워질수록 비어가며, 완전히 비면
 * 화면에서 사라집니다. 영구 분류가 아니라 임시 보관입니다.
 */
export const PENDING_CATEGORY_NAME = "아직 분류하지 않은 프로그램";

/**
 * 분야마다 "나머지"를 담을 묶음 이름입니다. 여기에 적힌 분야는 미분류 임시칸
 * 대신 이 이름의 정식 묶음이 생기고, 적지 않은 분야는 그대로 임시칸에
 * 남습니다. (2026-08-05 지정)
 */
export const FIELD_FALLBACK_CATEGORY: Record<string, string> = {
  transport: "행정 지원 및 기타 프로그램",
  agriculture: "행정지원 탄소중립기반",
  culture: "행정지원 기타",
  welfare: "일반복지 복권기금",
  industry: "행정 지원 기타",
  administration: "일반행정",
  environment: "행정지원 기타",
};

const FALLBACK_CATEGORY_DESCRIPTION =
  "위 묶음 어디에도 들어가지 않는 행정지원·기금운영 성격의 프로그램입니다.";

/**
 * 기존 묶음 이름 바꾸기. 바꾼 이름이 서로 같으면 한 묶음으로 합쳐집니다.
 * (2026-08-05: 통일·외교의 두 묶음을 "남북관계 통일정책"으로 통합)
 */
const GROUP_RENAMES: Record<string, Record<string, string>> = {
  diplomacy: {
    "남북관계·협력기금": "남북관계 통일정책",
    "통일정책·교육": "남북관계 통일정책",
  },
};

/**
 * 없앨 묶음. 여기 적힌 묶음은 사라지고, 속해 있던 프로그램은 그 분야의
 * "나머지" 묶음으로 내려갑니다.
 * (2026-08-05: 산업의 외국인투자유치를 "행정 지원 기타"로 흡수)
 *
 * 2026-08-07: 그 바람에 "외국인투자유치활성화"가 행정 항목처럼 보이게 되어,
 * program-groups-extra.ts에서 "산업기술·경쟁력"으로 옮겼습니다. 묶음 이름만
 * 없애고 프로그램은 제자리를 찾아 준 셈입니다.
 */
const DISSOLVED_GROUPS: Record<string, string[]> = {
  industry: ["외국인투자유치"],
};

const mergeGroups = (fieldId: string): ProgramGroup[] => {
  const renames = GROUP_RENAMES[fieldId] ?? {};
  const dissolved = new Set(DISSOLVED_GROUPS[fieldId] ?? []);
  const merged: ProgramGroup[] = [];
  for (const group of PROGRAM_GROUPS[fieldId] ?? []) {
    if (dissolved.has(group.name)) continue;
    const name = renames[group.name] ?? group.name;
    const existing = merged.find((candidate) => candidate.name === name);
    if (existing) {
      existing.programs.push(...group.programs);
      continue;
    }
    merged.push({ ...group, name, programs: [...group.programs] });
  }
  for (const extra of PROGRAM_GROUPS_EXTRA[fieldId] ?? []) {
    const existing = merged.find((group) => group.name === extra.name);
    if (existing) {
      existing.programs.push(...extra.programs);
      if (extra.description) existing.description = extra.description;
    } else {
      merged.push({ ...extra, programs: [...extra.programs] });
    }
  }
  return merged;
};

export const PROGRAM_CATEGORIES: ProgramCategory[] = FIELDS.flatMap((field) => {
  const fieldPrograms = SELECTABLE_PROGRAMS.filter(
    (program) => program.fieldId === field.id,
  );
  if (fieldPrograms.length === 0) return [];

  // 같은 분야 안에서 이름이 겹치는 프로그램이 있어 이름 하나에 id 여러 개를 답니다.
  //
  // 화면 이름(name)이 아니라 원본 이름(officialName)으로 답니다. 묶음 목록
  // (program-groups.ts)은 시트의 원본 이름으로 적혀 있어서, PROGRAM_RENAMES로
  // 이름을 바꾼 프로그램을 화면 이름으로 찾으면 어느 묶음에도 못 들어가고
  // "나머지" 칸으로 떨어집니다.
  const idsByName = new Map<string, string[]>();
  for (const program of fieldPrograms) {
    const bucket = idsByName.get(program.officialName);
    if (bucket) bucket.push(program.id);
    else idsByName.set(program.officialName, [program.id]);
  }

  const placed = new Set<string>();
  const categories: ProgramCategory[] = [];
  mergeGroups(field.id).forEach((group, index) => {
    const programIds = group.programs.flatMap((name) => idsByName.get(name) ?? []);
    if (programIds.length === 0) return;
    for (const id of programIds) placed.add(id);
    categories.push({
      id: `${field.id}-cat-${index + 1}`,
      fieldId: field.id,
      name: group.name,
      description: group.description,
      summary: firstSentence(group.description),
      programIds,
    });
  });

  const unplaced = fieldPrograms
    .filter((program) => !placed.has(program.id))
    .map((program) => program.id);
  const fallbackName = FIELD_FALLBACK_CATEGORY[field.id];
  if (unplaced.length > 0) {
    const description = fallbackName
      ? FALLBACK_CATEGORY_DESCRIPTION
      : "묶음 분류를 아직 마치지 않은 프로그램입니다. 분류가 끝나면 위 묶음들로 옮겨집니다.";
    categories.push({
      id: fallbackName ? `${field.id}-cat-etc` : `${field.id}-cat-pending`,
      fieldId: field.id,
      name: fallbackName ?? PENDING_CATEGORY_NAME,
      description,
      summary: firstSentence(description),
      programIds: unplaced,
      pending: fallbackName ? undefined : true,
    });
  }

  return categories;
});

export const CATEGORIES_BY_FIELD = new Map<string, ProgramCategory[]>();
for (const category of PROGRAM_CATEGORIES) {
  const bucket = CATEGORIES_BY_FIELD.get(category.fieldId);
  if (bucket) bucket.push(category);
  else CATEGORIES_BY_FIELD.set(category.fieldId, [category]);
}

export const CATEGORY_BY_PROGRAM_ID = new Map<string, ProgramCategory>();
for (const category of PROGRAM_CATEGORIES) {
  for (const programId of category.programIds) {
    CATEGORY_BY_PROGRAM_ID.set(programId, category);
  }
}

  return program.includedPrograms.join(", ");
}
