import {
  index,
  integer,
  sqliteTable,
  text,
  uniqueIndex,
} from "drizzle-orm/sqlite-core";

export const budgetResponses = sqliteTable(
  "budget_responses",
  {
    id: text("id").primaryKey(),
    sessionId: text("session_id").notNull().unique(),
    createdAt: text("created_at").notNull(),
    nickname: text("nickname").notNull(),
    ageGroup: text("age_group"),
    region: text("region"),
    selectedFieldsJson: text("selected_fields_json").notNull(),
    allocationJson: text("allocation_json").notNull(),
    customProposalJson: text("custom_proposal_json"),
    additionJson: text("addition_json"),
    increaseProposalJson: text("increase_proposal_json"),
    finalAllocationJson: text("final_allocation_json"),
    programDatasetVersion: text("program_dataset_version")
      .notNull()
      .default("budget-categories-v3-2026-08-03"),
    statedPriority: text("stated_priority"),
    consentVersion: text("consent_version").notNull(),
    researchConsent: integer("research_consent", { mode: "boolean" })
      .notNull()
      .default(false),
    completionMs: integer("completion_ms"),
    deleteTokenHash: text("delete_token_hash").notNull(),
    isTest: integer("is_test", { mode: "boolean" }).notNull().default(false),
  },
  (table) => [
    index("budget_responses_created_at_idx").on(table.createdAt),
    index("budget_responses_session_id_idx").on(table.sessionId),
  ],
);

export const budgetContacts = sqliteTable(
  "budget_contacts",
  {
    sessionId: text("session_id")
      .primaryKey()
      .references(() => budgetResponses.sessionId, { onDelete: "cascade" }),
    email: text("email").notNull(),
    createdAt: text("created_at").notNull(),
    /*
      소식 수신에 동의했는지와 그 시각.

      이메일은 응답을 구분하려고 받는 것이고 소식 수신은 별개의 동의라,
      두 가지를 한 칸에 뭉뚱그리면 나중에 "이 주소로 소식을 보내도 되는
      사람이었나"에 답할 수 없습니다. 그래서 이메일이 있다는 사실과
      동의했다는 사실을 따로 남깁니다.
    */
    marketingConsent: integer("marketing_consent", { mode: "boolean" })
      .notNull()
      .default(false),
    consentedAt: text("consented_at"),
  },
  (table) => [index("budget_contacts_email_idx").on(table.email)],
);

/**
 * 한 이메일당 한 번의 편성만 인정하기 위한 서버 측 잠금입니다.
 *
 * 기존 contact 행은 보존하면서 정규화된 이메일만 한 줄로 모읍니다. sessionId를
 * 함께 두는 이유는 같은 세션의 저장 재시도는 허용하고, 다른 세션만 중복으로
 * 막기 위해서입니다.
 */
export const participantEmailClaims = sqliteTable("participant_email_claims", {
  email: text("email").primaryKey(),
  sessionId: text("session_id").notNull().unique(),
  claimedAt: text("claimed_at").notNull(),
});

/**
 * 참여를 시작했지만 아직 결과를 저장하지 않은 흐름도 운영상 확인하기 위한 기록입니다.
 * 원문 이메일은 넣지 않고, 같은 이메일의 재시도 여부만 알 수 있는 가명 해시만 둡니다.
 */
export const participationAttempts = sqliteTable(
  "participation_attempts",
  {
    sessionId: text("session_id").primaryKey(),
    emailHash: text("email_hash").notNull(),
    startedAt: text("started_at").notNull(),
    lastSeenAt: text("last_seen_at").notNull(),
    lastEventName: text("last_event_name").notNull().default("started"),
    lastStep: integer("last_step").notNull().default(0),
    completedAt: text("completed_at"),
    entrySource: text("entry_source"),
    entryMedium: text("entry_medium"),
    entryCampaign: text("entry_campaign"),
    referrerHost: text("referrer_host"),
    deviceType: text("device_type"),
  },
  (table) => [
    index("participation_attempts_email_hash_idx").on(table.emailHash),
    index("participation_attempts_last_seen_at_idx").on(table.lastSeenAt),
    index("participation_attempts_completed_at_idx").on(table.completedAt),
  ],
);

/**
 * 친구 예산 케미의 한 판입니다.
 *
 * 초대한 사람과 초대받은 사람의 세션만 이어 둡니다. 케미 점수·칭호 같은
 * 판정 결과는 저장하지 않습니다. 두 사람의 배분표가 이미 budget_responses에
 * 있어 볼 때마다 다시 계산할 수 있고, 그래야 축 값(budget-axes.ts)을 나중에
 * 고쳤을 때 지난 대결도 새 값으로 읽힙니다. 점수를 컬럼에 박아 두면 옛 축에
 * 갇힌 결과만 남습니다.
 *
 * 두 세션 모두 budget_responses를 참조하므로, 어느 한쪽이 자기 응답을
 * 지우면 대결도 함께 사라집니다. "한 사람이 지우면 비교 결과도 닫는다"를
 * 애플리케이션 코드가 아니라 데이터베이스가 지키게 하는 편이 안전합니다.
 *
 * 친구마다 새 행을 만듭니다. 링크 하나를 여럿에게 돌리면 누구와의 결과인지
 * 알 수 없고, 먼저 연 사람이 자리를 차지해 버립니다.
 */
export const budgetDuels = sqliteTable(
  "budget_duels",
  {
    /** 초대 링크에 실리는 값입니다. 세션 id를 그대로 쓰지 않습니다. */
    id: text("id").primaryKey(),
    inviterSessionId: text("inviter_session_id")
      .notNull()
      .references(() => budgetResponses.sessionId, { onDelete: "cascade" }),
    /** 아직 아무도 응하지 않았으면 비어 있습니다. */
    inviteeSessionId: text("invitee_session_id").references(
      () => budgetResponses.sessionId,
      { onDelete: "cascade" },
    ),
    createdAt: text("created_at").notNull(),
    joinedAt: text("joined_at"),
    /** 기본 30일. 지나면 결과를 열지 않습니다. */
    expiresAt: text("expires_at").notNull(),
  },
  (table) => [
    index("budget_duels_inviter_idx").on(table.inviterSessionId),
    index("budget_duels_invitee_idx").on(table.inviteeSessionId),
    index("budget_duels_created_at_idx").on(table.createdAt),
  ],
);

    statedPriority: text("stated_priority"),
    selectedFieldsJson: text("selected_fields_json").notNull(),
    initialAllocationJson: text("initial_allocation_json").notNull(),
    customProposalJson: text("custom_proposal_json"),
    cutsJson: text("cuts_json").notNull(),
    additionJson: text("addition_json").notNull(),
    increaseProposalJson: text("increase_proposal_json"),
    finalAllocationJson: text("final_allocation_json").notNull(),

    isTest: integer("is_test", { mode: "boolean" }).notNull().default(false),

export const budgetProposals = sqliteTable(
  "budget_proposals",
  {
    id: text("id").primaryKey(),
    sessionId: text("session_id").notNull(),
    round: text("round").notNull(),
    fieldId: text("field_id").notNull(),
    fieldName: text("field_name").notNull(),
    title: text("title").notNull(),
    description: text("description"),
    amount: integer("amount").notNull(),
    createdAt: text("created_at").notNull(),
  },
  (table) => [
    uniqueIndex("budget_proposals_session_round_unique").on(
      table.sessionId,
      table.round,
    ),
    index("budget_proposals_round_field_idx").on(table.round, table.fieldId),
    index("budget_proposals_created_at_idx").on(table.createdAt),
  ],
);

export const contacts = sqliteTable(
  "contacts",
  {
    id: text("id").primaryKey(),
    responseId: text("response_id")
      .notNull()
      .unique()
      .references(() => responses.id, { onDelete: "cascade" }),
    email: text("email").notNull(),
    marketingConsent: integer("marketing_consent", { mode: "boolean" })
      .notNull()
      .default(false),
    consentedAt: text("consented_at").notNull(),
  },
  (table) => [index("contacts_consent_idx").on(table.marketingConsent)],
);

export const validationResponses = sqliteTable("validation_responses", {
  responseId: text("response_id")
    .primaryKey()
    .references(() => responses.id, { onDelete: "cascade" }),
  answersJson: text("answers_json").notNull(),
  responseMetaJson: text("response_meta_json").notNull().default("{}"),
  selfFuture: integer("self_future").notNull(),
  selfGrowth: integer("self_growth").notNull(),
  selfUniversal: integer("self_universal").notNull(),
  agreement: integer("agreement").notNull(),
  accuracyRating: integer("accuracy_rating"),
  feedbackComment: text("feedback_comment"),
  feedbackAt: text("feedback_at"),
});

export const events = sqliteTable(
  "events",
  {
    id: integer("id").primaryKey({ autoIncrement: true }),
    sessionId: text("session_id").notNull(),
    eventName: text("event_name").notNull(),
    step: integer("step"),
    payloadJson: text("payload_json"),
    occurredAt: text("occurred_at").notNull(),
  },
  (table) => [
    index("events_session_id_idx").on(table.sessionId),
    index("events_name_idx").on(table.eventName),
    index("events_occurred_at_idx").on(table.occurredAt),
  ],
);

export type ResponseRecord = typeof responses.$inferSelect;
